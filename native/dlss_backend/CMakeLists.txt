cmake_minimum_required(VERSION 3.24)
project(studio_backend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/color.hlsl" STUDIO_SHADER)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shader.inc" "R\"HLSL(${STUDIO_SHADER})HLSL\"")
add_library(studio_backend STATIC src/backend.cpp)
target_compile_definitions(studio_backend PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(studio_backend PUBLIC include PRIVATE ../../third_party/streamline/include "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(studio_backend PUBLIC d3d12 dxgi d3dcompiler wintrust crypt32 version)
install(TARGETS studio_backend ARCHIVE DESTINATION lib)
add_executable(native_smoke tests/smoke.cpp)
target_link_libraries(native_smoke PRIVATE studio_backend)
enable_testing()
add_test(NAME native_smoke COMMAND native_smoke)
